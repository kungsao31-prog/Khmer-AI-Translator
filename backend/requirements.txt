import os
import tempfile
from flask import Flask, request, jsonify
from google import genai

app = Flask(__name__)

# ទាញយក API Key ពី Environment Variables របស់ Render
GEMINI_API_KEY = os.environ.get("GEMINI_API_KEY")

# ផ្ទៀងផ្ទាត់ការតភ្ជាប់ Gemini Client
client = genai.Client(api_key=GEMINI_API_KEY) if GEMINI_API_KEY else None

@app.route('/', methods=['GET'])
def health_check():
    """Route សម្រាប់ពិនិត្យស្ថានភាព Server និង Render Health Check"""
    return jsonify({
        "status": "online",
        "message": "Khmer AI Translator Backend is running successfully!",
        "gemini_configured": client is not None
    }), 200

@app.route('/translate-video', methods=['POST'])
def translate_video():
    """Endpoint សម្រាប់ទទួលសំណើបកប្រែវីដេអូ"""
    if not client:
        return jsonify({"error": "Gemini API Key is missing on the server backend."}), 500
    
    try:
        video_file = request.files.get('video')
        data = request.get_json(silent=True) or {}
        video_url = data.get('video_url')

        if not video_file and not video_url:
            return jsonify({"error": "សូមបញ្ជូន File វីដេអូ ឬ Video URL មកជាមួយ។"}), 400

        # ករណីផ្ញើមកជា File វីដេអូផ្ទាល់
        if video_file:
            with tempfile.NamedTemporaryFile(delete=False, suffix='.mp4') as temp_vid:
                video_file.save(temp_vid.name)
                temp_vid_path = temp_vid.name

            # Upload ទៅកាន់ Gemini Files API
            uploaded_file = client.files.upload(file=temp_vid_path)

            prompt = (
                "សូមស្តាប់វីដេអូនេះ បកប្រែអត្ថន័យជាភាសាខ្មែរ និងបង្កើតជាទម្រង់អត្ថបទរង SRT "
                "(មានម៉ោងចាប់ផ្តើម និងបញ្ចប់ ព្រមទាំងអត្ថបទបកប្រែជាភាសាខ្មែរ)។"
            )
            
            response = client.models.generate_content(
                model='gemini-2.5-flash',
                contents=[uploaded_file, prompt]
            )

            # លុបឯកសារបណ្តោះអាសន្ន
            os.unlink(temp_vid_path)
            client.files.delete(name=uploaded_file.name)

            return jsonify({
                "success": True,
                "translation_result": response.text
            }), 200

        # ករណីផ្ញើមកជា Video URL
        elif video_url:
            prompt = (
                f"សូមវិភាគ និងបកប្រែវីដេអូពីតំណភ្ជាប់នេះ (URL: {video_url}) ទៅជាភាសាខ្មែរ "
                "ព្រមទាំងបង្កើតជាទម្រង់ SRT Subtitle ជូនខ្ញុំផង។"
            )
            response = client.models.generate_content(
                model='gemini-2.5-flash',
                contents=prompt
            )
            return jsonify({
                "success": True,
                "translation_result": response.text
            }), 200

    except Exception as e:
        return jsonify({"error": str(e)}), 500

if __name__ == '__main__':
    port = int(os.environ.get("PORT", 5000))
    app.run(host='0.0.0.0', port=port)
